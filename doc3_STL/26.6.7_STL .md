案例
```
有5名选手:选手ABCDE，10个评委分别对每一名选手打分，去除最高分，去除评委中最低分，取平均分。
1.创建五名选手，放到vector中
2.遍历vector 容器，取出来每一个选手，执行for循环，可以把10个评分打分存到deque容器中
3.sort算法对deque容器中分数排序，pop_back pop_front去除最高和最低分
4.deque容器遍历一遍，累加分数，累加分数/d.size()
5.person.score=平均分

#include <iostream>
#include <deque>
#include <algorithm>
using namespace std;

class Person
{
public:
    string name;
    float score;
public:
    Person(){};
    Person(string name, float score)
    {
        this->name = name;
        this->score = score;
    
    }
};

void createPerson(vector<Person> &v)     //创建选手
{
    string tmpName = "ABCDE";
    int i = 0;
    for(i=0;i<5;i++)
    {
        string name = "选手";
        name += tmpName[i];
        v.push_back(Person(name, 0.0f))
    }
    
}

void showPerson(vector<Person> &v)      //展示选手
{
    vector<Person>::iterator it = v.begin();
    for(;it!=v.end();it++)
    {
        //*it == Person
        cout<<(*it).name<<" "<<(*it).score<<endl;
    }
}

void playGame(vector<Person> &v)
{
    //每个人都参加比赛  用迭代器
    vector<Person> :: iterator it = v.begin();
    for(;it!=v.end();it++)
    {
        //定义一个deque容器存放10个评委的分数
        deque<float> d;
        int i = 0;
        for(i=0;i<10;i++)     //10个评委打分
        {
             d.push_back( (float)(rand()%41+60) )   //   (0~40)+60 保证分数在60-100
        }
        //对deque排序
        sort(d.begin(), d.end());
        //去掉最高分和最低分
        d.pop_back();
        d.pop_front();
        //求平局分
        (*it).score = (float)accumulate(d.begin(), d.end(), 0)/d.size(); 

    } 
}
#include<time>

void test02()
{
    vector<Person> v;

    //srand设置随机数种子
    srand(time(NULL));
    //创建5名选手
    createPerson(v);
    //显示选手成绩
    showPerson(v);
    //比赛
    playGame(v)
}

int main(int argc, char* argv[])
{
    test02();
    return 0;
}


```

## 5.list 链表容器     -----双向链表
```
链表是一种物理存储单元上非连续、非顺序的存储结构，数据元素的逻辑顺序是通过链表中的指针连接次序实现的
链表由一系列结点(链表中每一个元素称之为结点)组成，结点可以在运行时动态生成

结点两部分：
1)存储数据元素的数据域
2)存储下一个结点地址的指针域

list:好处  负责任：
每次插入或删除一个元素就是配置或释放一个元素空间 -----list对于空间的运用绝对精准 不浪费

采用动态存储分配，不会造成内存浪费和溢出，链表执行插入和删除十分方便，修改指针即可，不需要移动大量数据
但是！！!  空间和时间额外消耗大

```
