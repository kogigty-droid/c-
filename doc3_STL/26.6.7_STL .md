3.vector 插入和删除操作

5.vector 容器提高
  嵌套容器定义一个容器存放v1 v2 v3

  ```c++
  vector<int> v1(5,10);
  vector<int> v2(5,100);
  vector<int> v3(5,1000);
  //需求: 定义一个容器存放v1 v2 v3
  vector< vector<int> > v;
  v.push_back(v1);
  v.push_back(v2);
  v.push_back(v3);

  //遍历要定义他的迭代器
  vector< vector<int> > :: iterator it = v.begin()
  for(;it!=v.end();it++)
  {
     //*it==vector<int>  **对迭代器取*代表的就是尖括号里面的内容vector<int>**
     //容器嵌套最关键的就是 这个*it到底是什么！！！
     vector<int>::iterator mit = (*it).begin();
     for(; mit!=(*it).end(); (*it)++ )
     {
         cout<<*mit<<"";
     }
         cout<<endl;

  }
